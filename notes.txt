# ssh to debian1
ssh debian@192.168.128.2

#create and give IP to kali on the private network (only has IP on the shared one if two interfaces)
automatic from sudo vim /etc/network/interfaces



#set syncookies on debian to 0
# On Debian set securities off, on rp_filter - filter both - kernel takes the stricter with 0
sudo sysctl -w net.ipv4.tcp_syncookies=0

#NOT NEEDED
sudo sysctl -w net.ipv4.conf.all.rp_filter=0 
sudo sysctl -w net.ipv4.conf.enp0s1.rp_filter=0
sudo sysctl -w net.ipv4.tcp_max_syn_backlog=128

# block RST packets outgoing from Kali - otherwise kernel sends RST packets in response to SYN-ACKs destroying the purpose of half-open connections
sudo iptables -A OUTPUT -p tcp --tcp-flags RST RST -d 192.168.128.2 -j DROP

# view the TCP communication
sudo tcpdump -i enp0s1 'tcp port 80' -c 20
